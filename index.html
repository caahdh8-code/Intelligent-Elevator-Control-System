"message user-message" : "message ai-message"; div.textContent = message; box.appendChild(div); box.scrollTop = box.scrollHeight; } // ====================================== // TEXT NORMALISATION // ====================================== function normaliseText(value) { return String(value || "") .toLowerCase() .normalize("NFD") .replace(/[\u0300-\u036f]/g, "") .replace(/[’']/g, "") .replace(/[.,!?;:()]/g, " ") .replace(/\s+/g, " ") .trim(); const numberPatterns = [ /\b(?:saya|aku|awak|i am|im|i'm)\s+(?:sekarang\s+)?(?:berada\s+)?(?:di|dekat|dari|from|on|at)\s+(?:floor|level|tingkat)?\s*([1-3])\b/, /\b(?:saya|aku|i am|im|i'm)\s+(?:datang|bermula)\s+dari\s+(?:floor|level|tingkat)\s*([1-3])\b/, /\b(?:dari|from)\s+(?:floor|level|tingkat)\s*([1-3])\b/, /\b(?:current floor|my floor|floor saya|tingkat saya)\s*(?:is|ialah|adalah)?\s*([1-3])\b/ ]; for (const pattern of numberPatterns) { const match = text.match(pattern); if (match) return Number(match[1]); } const words = { satu: 1, pertama: 1, one: 1, dua: 2, kedua: 2, two: 2, tiga: 3, ketiga: 3, three: 3 }; const wordPattern = /\b(?:saya|aku|i am|im|i'm)\s+(?:sekarang\s+)?(?:berada\s+)?(?:di|dekat|dari|from|on|at)\s+(?:floor|level|tingkat)?\s*(satu|pertama|one|dua|kedua|two|tiga|ketiga|three)\b/; const wordMatch = text.match(wordPattern); if (shortMatch) return words[shortMatch[1]] || Number(shortMatch[1]); } return null; } // ====================================== // CONFIRMATION DETECTION // ====================================== function isConfirmation(message) { const text = expandMalayShortform(message); return [ "yes", "yeah", "yep", "ya", "ye", "ok", "okay", "baik", "baiklah", "betul", "setuju", "sahkan", "confirm", "boleh", "teruskan", "hantar", "terus", "haah", "aah", "sure", "go ahead", "do it", "jom", "nak" ].includes(text); } function isRejection(message) { const text = expandMalayShortform(message); return [ "no", "nope", "cancel", "cancel it", "tidak", "batal", "jangan", "tak setuju", "tidak setuju", "tak nak", "tidak mahu", "tak jadi", "stop" ].includes(text); } // ====================================== // SEND AI REQUEST TO FIREBASE // ====================================== async function submitAILiftRequest(destination) { destination = Number(destination); if (!currentFloor) { waitingForCurrentFloor = true; addMessage( "Awak berada di tingkat berapa sekarang? " + error.message, "ai" ); } } // ====================================== // SEND MESSAGE TO AI // ====================================== async function sendMessage() { if (isSendingMessage) return; const input = document.getElementById("chatInput"); const originalText = input.value.trim(); if (!originalText) return; input.value = ""; addMessage(originalText, "user"); // -------------------------------------- // 1. Detect current floor from message // -------------------------------------- const detectedFloor = detectCurrentFloor(originalText); if (detectedFloor !== null) { updateCurrentFloor(detectedFloor); const wasWaiting = waitingForCurrentFloor; waitingForCurrentFloor = false; // Direct travel: once current floor is supplied, // send immediately without asking again. if (pendingLiftRequest && pendingLiftRequest.type === "direct" && wasWaiting) { const destination = pendingLiftRequest.destinationFloor; pendingLiftRequest = null; await submitAILiftRequest(destination); return; } // Location question: after current floor is supplied, // ask for confirmation before sending. if (pendingLiftRequest && pendingLiftRequest.type === "offer" && wasWaiting) { const destination = pendingLiftRequest.destinationFloor; if (currentFloor === destination) { pendingLiftRequest = null; addMessage("Awak sudah berada di tingkat ini!", "ai"); return; } addMessage( "Baik, awak sekarang di Tingkat " + currentFloor + ". Direct command missing current floor // -------------------------------------- if (data.action === "ask_current_floor") { pendingLiftRequest = { type: "direct", destinationFloor: Number(data.destinationFloor) }; waitingForCurrentFloor = true; addMessage(reply, "ai"); return; } // -------------------------------------- // 6. Normal AI conversation // -------------------------------------- addMessage(reply, "ai"); } catch (error) { thinking.remove(); console.error("AI error:", error); addMessage( "Maaf, AI Assistant tidak dapat dihubungi buat masa ini.\n" + error.message, "ai" ); } finally { isSendingMessage = false; document.getElementById("sendButton").disabled = false; } } // ====================================== // VOICE INPUT // ====================================== function startVoice() { const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition; const status = document.getElementById("voiceStatus"); const voiceButton = document.getElementById("voiceButton"); const selectedLanguage = document.getElementById("voiceLanguage").value; if (!SpeechRecognition) { status.textContent = "Voice recognition is not supported.
